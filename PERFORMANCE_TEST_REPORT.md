## 📦 12. Recommended Jenkins Pipeline Structure

```groovy
pipeline {
    agent any

    environment {
        JMETER_HOME = 'C:\\apache-jmeter-5.6.3'
        RESULTS_DIR = 'results'
        REPORT_DIR  = 'results\\report'
    }

    stages {

        stage('Checkout') {
            steps {
                git branch: 'main', url: 'https://github.com/nirmaldangal/performance_jenkins'
            }
        }

        stage('Clean Workspace') {
            steps {
                bat 'if exist %RESULTS_DIR% rmdir /s /q %RESULTS_DIR%'
                bat 'mkdir %RESULTS_DIR%'
            }
        }

        stage('Run JMeter Test') {
            steps {
                bat '''
                %JMETER_HOME%\\bin\\jmeter.bat -n ^
                  -t practice.jmx ^
                  -l %RESULTS_DIR%\\results.jtl ^
                  -e -o %REPORT_DIR% ^
                  -Jthreads=10 ^
                  -Jrampup=5 ^
                  -Jduration=60
                '''
            }
        }

        stage('Generate Markdown Report') {
            steps {
                bat 'python generate_report.py'
            }
        }

        stage('Publish HTML Report') {
            steps {
                publishHTML(target: [
                    allowMissing         : false,
                    alwaysLinkToLastBuild: true,
                    keepAll              : true,
                    reportDir            : "${REPORT_DIR}",
                    reportFiles          : 'index.html',
                    reportName           : 'Performance Report'
                ])
            }
        }

        stage('Archive Artifacts') {
            steps {
                archiveArtifacts artifacts: 'results/results.jtl, performance-report.md', fingerprint: true
            }
        }

        stage('Send Email Report') {
            steps {
                emailext(
                    subject: "Performance Test Report — Build #${BUILD_NUMBER} — ${currentBuild.currentResult}",
                    body: """
                        <h2>Performance Test Completed</h2>
                        <p><b>Job:</b> ${JOB_NAME}</p>
                        <p><b>Build:</b> #${BUILD_NUMBER}</p>
                        <p><b>Status:</b> ${currentBuild.currentResult}</p>
                        <p><b>HTML Report:</b> <a href="${BUILD_URL}Performance_Report/">View Report</a></p>
                        <p><b>Build URL:</b> <a href="${BUILD_URL}">${BUILD_URL}</a></p>
                    """,
                    mimeType: 'text/html',
                    to: 'it.nirmaldangal@gmail.com',
                    attachmentsPattern: 'performance-report.md, results/results.jtl'
                )
            }
        }
    }

    post {
        always {
            echo "✅ Pipeline completed. Status: ${currentBuild.currentResult}"
        }
        failure {
            echo "❌ Build failed — check Console Output for details."
        }
    }
}
```
